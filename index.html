DATABASE & MEMORY STORAGE (SQLite3) # ========================================== class DatabaseManager: def __init__(self, db_name="parnarav_ai.db"): self.db_name = db_name self.init_db() def get_connection(self): return sqlite3.connect(self.db_name) def init_db(self): with self.get_connection() as conn: cursor = conn.cursor() # Chat history table cursor.execute(''' CREATE TABLE IF NOT EXISTS chat_history ( id INTEGER PRIMARY KEY AUTOINCREMENT, sender TEXT NOT NULL, message TEXT NOT NULL, timestamp DATETIME DEFAULT CURRENT_TIMESTAMP, mode TEXT, is_captain INTEGER DEFAULT 0 ) ''') # Activity audit table cursor.execute(''' CREATE TABLE IF NOT EXISTS activity_logs ( id INTEGER PRIMARY KEY AUTOINCREMENT, action_type TEXT NOT NULL, details TEXT, timestamp DATETIME DEFAULT CURRENT_TIMESTAMP ) ''') conn.commit() def save_chat(self, sender, message, mode="OMNI", is_captain=False): with self.get_connection() as conn: cursor = conn.cursor() cursor.execute( "INSERT INTO chat_history (sender, message, mode, is_captain) VALUES (?, ?, ?, ?)", (sender, message, mode, 1 if is_captain else 0) ) conn.commit() def log_activity(self, action_type, details=""): with self.get_connection() as conn: cursor = conn.cursor() cursor.execute( "INSERT INTO activity_logs (action_type, details) VALUES (?, ?)", (action_type, details) ) conn.commit() def fetch_recent_chats(self, limit=20): with self.get_connection() as conn: cursor = conn.cursor() cursor.execute("SELECT sender, message, timestamp, mode FROM chat_history ORDER BY id DESC LIMIT ?", (limit,)) return cursor.fetchall() def fetch_all_activities(self): with self.get_connection() as conn: cursor = conn.cursor() cursor.execute("SELECT action_type, details, timestamp FROM activity_logs ORDER BY id DESC LIMIT 50") return cursor.fetchall() def clear_database(self): with self.get_connection() as conn: cursor = conn.cursor() cursor.execute("DELETE FROM chat_history") cursor.execute("DELETE FROM activity_logs") conn.commit() # BUILT-IN CALCULATORS & ENGINES # ========================================== class QuantEngine: @staticmethod def generate_analysis(symbol="BTC/USD"): rsi = round(random.uniform(22.0, 78.0), 2) macd = round(random.uniform(-15.0, 25.0), 2) signal = "STRONG BUY 🚀" if rsi < 35 else ("STRONG SELL 📉" if rsi > 65 else "NEUTRAL ⚡") report = f"""📊 [REAL-TIME QUANT INDICATOR] • Symbol Pair: {symbol} • RSI (14 Period): {rsi} • MACD Signal Line: {macd} • Moving Average: Bullish Crossover (50/200) • Final Verdict: {signal} • Confidence Index: {random.randint(85, 98)}%""" return report class FinanceEngine: @staticmethod def calculate_sip(monthly_investment, return_rate_annual, years): try: r = (return_rate_annual / 100) / 12 n = years * 12 invested = monthly_investment * n future_value = monthly_investment * (((1 + r)**n - 1) / r) * (1 + r) wealth_gained = future_value - invested return { "invested": round(invested, 2), "future_value": round(future_value, 2), "wealth_gained": round(wealth_gained, 2) } except Exception: return None @staticmethod def calculate_emi(principal, annual_rate, years): try: r = (annual_rate / 12) / 100 n = years * 12 emi = (principal * r * (1 + r)**n) / ((1 + r)**n - 1) total_payment = emi * n total_interest = total_payment - principal return { "monthly_emi": round(emi, 2), "total_interest": round(total_interest, 2), "total_payment": round(total_payment, 2) } except Exception: return None # ========================================== # 3.
